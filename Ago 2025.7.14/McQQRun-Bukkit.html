主配置 config.yml

ws: "ws://127.0.0.1:3001"

BungeeCord: false   # true则不使用白名单功能(仅使用除绑定之类的功能)   false则使用

KickMessage: |
  您已被踢出服务器。
  请联系管理员了解详情。
  谢谢！

Groups:
  - "288526441"

群管:
  入群欢迎:
    开关: false
    提示词: "%艾特% 欢迎"
  退群解绑:
    开关: false
  进群验证:
    全局验证码过期时间: 5 # 单位 分钟 整数
    固定:
      开关: false
      验证码: "我不是机器人"
      提示词: "请在 %time% 分钟内输入：验证 %code% 以完成验证，否则将被移出群聊。"
    随机:
      开关: false
      提示词: "请在 %time% 分钟内输入验证码：验证 %code% ，否则将被移出群聊。"
      # 数字+字母组合的6位验证码
  验证:
    开关: false
    指令: "验证 "

主人:
  QQ:
    - "2139145308"
  不是主人: "你想怎样"

设置群员名称:
  开关: false
  格式: "超级帅的%id%" # %id% = 玩家ID

绑定:
  开关: true
  指令: "绑定 "
  提示: "%艾特% ID绑定成功"
  QQ重复: "您的QQ已在白名单中，如ID错误，请联系管理员进行处理"
  ID重复: "您的ID已在白名单中！"
  查看绑定: "我的绑定"
  查看: "您所绑定的ID是: %ID%\\n您的QQ: %QQ%"

  查看他人: true
  查看他人命令: "他人绑定 "

删除ID:
  开关: true
  指令: "删除ID "
  不存在: "该ID不存在"
  提示: "该ID已被删除"

服务器状态:
  开关: true
  是否允许玩家使用: true
  指令: "服务器状态"
  format: "%TPS%\\n在线玩家: %player_format%\\n%player_all%"
  nullPlayer: "    无人"

删除QQ:
  开关: true
  指令: "删除QQ "
  提示: "该QQ已被删除"
  不存在: "该QQ不存在"

Rcon:
  IP: "127.0.0.1"
  Port: 25566
  Password: "a"

控制台:
  开关: true
  使用Rcon: true
  指令: "server "

菜单:
  开关: true
  指令: "菜单"
  菜单: |
    请前往mcqqrun/config.yml修改

公告 Annoucement.yml

开关: false
公告:
  "1":
    time: 1500  #1000 = 1s
    内容: "1.5s发送一次公告"
  "abc":
    time: 11500  #1000 = 1s
    内容: "11.5s发送一次公告"
  "aaaaa":
    time: 1  #1000 = 1s
    内容: "11aaaas发送一次公告"

自定义回复 reply.yml

"1":
  message: "菜单"
  reply: "测试信息-菜单"
  exact: false #exact是准确的意思
"2":
  message: "你好"
  reply: "你好呀"
  exact: true
